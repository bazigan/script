import boto3

def delete_all_buckets():
    s3 = boto3.resource('s3')
    
    print("Memindai akun untuk mencari semua S3 bucket...")
    
    # Langsung mengambil semua bucket yang ada di akun
    semua_bucket = list(s3.buckets.all())
    
    if not semua_bucket:
        print("Tidak ada bucket yang ditemukan di akun ini. Selesai.")
        return

    print(f"\n⚠️ DITEMUKAN {len(semua_bucket)} BUCKET TOTAL ⚠️")
    for b in semua_bucket:
        print(f" - {b.name}")
        
    # Jaring pengaman terakhir
    konfirmasi = input("\nAPAKAH ANDA YAKIN INGIN MENGHAPUS SEMUA BUCKET DI ATAS SECARA PERMANEN? (ketik 'ya' / 'tidak'): ")
    
    if konfirmasi.lower() == 'ya':
        for bucket in semua_bucket:
            print(f"\n--- Menghancurkan: {bucket.name} ---")
            try:
                print("1. Mengosongkan semua data dan versi...")
                bucket.object_versions.delete()
                
                print("2. Menghapus bucket...")
                bucket.delete()
                print(f"✅ '{bucket.name}' hangus tak bersisa.")
            except Exception as e:
                print(f"❌ Gagal menghapus '{bucket.name}'. Error: {e}")
        print("\nProses sapu bersih selesai!")
    else:
        print("Proses dibatalkan. Data Anda aman.")

if __name__ == '__main__':
    delete_all_buckets()
